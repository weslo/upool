== ~/.upool-scope-capture/hook.sh, every hook's command in the scratch repository
#!/bin/sh
# Scope capture (#172): logs each hook run with millisecond times and its parent, then runs
# upool's hook for the worktree events and a marker export for SessionStart.
here="$HOME/.upool-scope-capture"
upool="${home}/Documents/Personal/upool/bin/upool"
ms() { /usr/bin/perl -MTime::HiRes=time -MPOSIX=strftime -e '$t=time; printf "%s.%03d", strftime("%Y-%m-%d %H:%M:%S", localtime $t), ($t-int $t)*1000'; }
event="$1"; src="$2"
stdin=$(cat)
parent=$(ps -o command= -p "$PPID" 2>/dev/null | cut -c1-100)
echo "$(ms) start $event from=$src ppid=$PPID parent=$parent" >>"$here/hooks.log"
printf '%s\n' "$stdin" >>"$here/stdin-$event.jsonl"
status=0; out=
case "$event" in
WorktreeCreate) out=$(printf '%s\n' "$stdin" | "$upool" hook worktree-create 2>>"$here/upool.stderr"); status=$? ;;
WorktreeRemove) out=$(printf '%s\n' "$stdin" | "$upool" hook worktree-remove 2>>"$here/upool.stderr"); status=$? ;;
SessionStart) if [ -n "${CLAUDE_ENV_FILE:-}" ]; then echo "export UPOOL_PROBE=scope-$src" >>"$CLAUDE_ENV_FILE"; fi ;;
esac
echo "$(ms) end $event from=$src status=$status" >>"$here/hooks.log"
if [ -n "$out" ]; then printf '%s\n' "$out"; fi
exit $status
