# The probe behind contracts.md A32 (#33), 2026-10-21, macOS 15.7.3 on APFS.
# privsize.c:

// privsize prints each path's APFS private size: the bytes of the file that no clone shares,
// from getattrlist's ATTR_CMNEXT_PRIVATESIZE.
#include <stdio.h>
#include <string.h>
#include <sys/attr.h>
#include <unistd.h>

struct reply {
	u_int32_t length;
	off_t private_size;
} __attribute__((aligned(4), packed));

int main(int argc, char **argv) {
	for (int i = 1; i < argc; i++) {
		struct attrlist req;
		memset(&req, 0, sizeof req);
		req.bitmapcount = ATTR_BIT_MAP_COUNT;
		req.forkattr = ATTR_CMNEXT_PRIVATESIZE;
		struct reply r;
		if (getattrlist(argv[i], &req, &r, sizeof r, FSOPT_ATTR_CMN_EXTENDED) != 0) {
			perror(argv[i]);
			return 1;
		}
		printf("%s private=%lld\n", argv[i], (long long)r.private_size);
	}
	return 0;
}

# Run in an empty directory on the boot volume:

$ clang -o privsize privsize.c
$ dd if=/dev/urandom of=a bs=1m count=20
$ ./privsize a
a private=20971520
$ cp -c a b && ./privsize a b
a private=0
b private=0
$ du -k a b
20480	a
20480	b
$ dd if=/dev/urandom of=b bs=1m count=2 conv=notrunc && ./privsize a b
a private=2097152
b private=2097152
$ du -k a b
20480	a
20480	b
$ rm b && ./privsize a
a private=20971520
