== ~/.upool-scope-probe/session-start.sh, the project's SessionStart hook
#!/bin/sh
# Scope probe: logs each SessionStart stdin with the hook's parent PID, and exports a marker.
here="$HOME/.upool-scope-probe"
{ printf '%s ppid=%s ' "$(date '+%Y-%m-%d %H:%M:%S')" "$PPID"; cat; echo; } >>"$here/session-start.log"
if [ -n "${CLAUDE_ENV_FILE:-}" ]; then echo 'export UPOOL_PROBE=project-scope' >>"$CLAUDE_ENV_FILE"; fi
exit 0
